📋 COPY-PASTE COMMANDS FOR PRODUCTION MIGRATION

═══════════════════════════════════════════════════════════════════════════════
STEP 1: CREATE CLOUD SQL INSTANCE (15 minutes)
═══════════════════════════════════════════════════════════════════════════════

# Set variables
$PROJECT_ID = "exalted-skein-505210-g0"
$INSTANCE_NAME = "coldsense-production-db"
$REGION = "asia-south1"
$ROOT_PASSWORD = "$(New-Guid)"
Write-Output "Root Password: $ROOT_PASSWORD" | Out-File "cloud-sql-credentials.txt"

# Create instance with High Availability
gcloud sql instances create $INSTANCE_NAME --project=$PROJECT_ID --database-version=POSTGRES_15 --tier=db-custom-2-7680 --region=$REGION --availability-type=REGIONAL --storage-type=SSD --storage-size=100GB --storage-auto-increase --storage-auto-increase-limit=500 --backup --backup-start-time="02:00" --maintenance-window-day=SUN --maintenance-window-hour=2 --enable-bin-log --deletion-protection --root-password="$ROOT_PASSWORD"

# ⏳ Wait 10-15 minutes for instance creation...

# Verify creation
gcloud sql instances describe $INSTANCE_NAME --project=$PROJECT_ID

═══════════════════════════════════════════════════════════════════════════════
STEP 2: CREATE DATABASE & USER (5 minutes)
═══════════════════════════════════════════════════════════════════════════════

# Create database
gcloud sql databases create coldsense --instance=$INSTANCE_NAME --project=$PROJECT_ID

# Create app user
$APP_PASSWORD = "$(New-Guid)"
Write-Output "App User Password: $APP_PASSWORD" | Out-File -Append "cloud-sql-credentials.txt"
gcloud sql users create coldsense_app --instance=$INSTANCE_NAME --password="$APP_PASSWORD" --project=$PROJECT_ID

# Get connection details
$CONNECTION_NAME = gcloud sql instances describe $INSTANCE_NAME --project=$PROJECT_ID --format="get(connectionName)"
$PUBLIC_IP = gcloud sql instances describe $INSTANCE_NAME --project=$PROJECT_ID --format="get(ipAddresses[0].ipAddress)"

Write-Host "Connection Name: $CONNECTION_NAME"
Write-Host "Public IP: $PUBLIC_IP"
Write-Host "Database: coldsense"
Write-Host "User: coldsense_app"

# Save connection string
$DB_URL = "postgresql://coldsense_app:$APP_PASSWORD@$PUBLIC_IP:5432/coldsense?sslmode=require"
Write-Output "Database URL: $DB_URL" | Out-File -Append "cloud-sql-credentials.txt"

═══════════════════════════════════════════════════════════════════════════════
STEP 3: CONFIGURE SECURITY (5 minutes)
═══════════════════════════════════════════════════════════════════════════════

# Get backend VM IP
$BACKEND_VM_IP = gcloud compute instances describe coldsense-backend-vm --zone=asia-south1-a --format="get(networkInterfaces[0].accessConfigs[0].natIP)"
Write-Host "Backend VM IP: $BACKEND_VM_IP"

# Allow backend VM to connect
gcloud sql instances patch $INSTANCE_NAME --authorized-networks="$BACKEND_VM_IP/32" --project=$PROJECT_ID

# Require SSL
gcloud sql instances patch $INSTANCE_NAME --require-ssl --project=$PROJECT_ID

═══════════════════════════════════════════════════════════════════════════════
STEP 4: EXPORT FROM SUPABASE (20 minutes)
═══════════════════════════════════════════════════════════════════════════════

# SSH to backend VM
gcloud compute ssh coldsense-backend-vm --zone=asia-south1-a

# === ON THE VM (after SSH) ===

# Install postgresql-client
sudo apt-get update && sudo apt-get install -y postgresql-client

# REPLACE WITH YOUR SUPABASE CONNECTION STRING
export SUPABASE_URL="postgresql://postgres.mmxfqodybvdlhpjbcdnq:YOUR_PASSWORD@aws-0-ap-south-1.pooler.supabase.com:6543/postgres"

# Export schema
pg_dump "$SUPABASE_URL" --schema-only --no-owner --no-acl > /tmp/coldsense_schema.sql

# Export data
pg_dump "$SUPABASE_URL" --data-only --no-owner --no-acl --column-inserts > /tmp/coldsense_data.sql

# Exit VM
exit

# === BACK ON LOCAL MACHINE ===

# Download exports
gcloud compute scp coldsense-backend-vm:/tmp/coldsense_schema.sql . --zone=asia-south1-a
gcloud compute scp coldsense-backend-vm:/tmp/coldsense_data.sql . --zone=asia-south1-a

═══════════════════════════════════════════════════════════════════════════════
STEP 5: IMPORT TO CLOUD SQL (30 minutes)
═══════════════════════════════════════════════════════════════════════════════

# Create Cloud Storage bucket
$BUCKET_NAME = "$PROJECT_ID-sql-import"
gsutil mb -p $PROJECT_ID -l $REGION gs://$BUCKET_NAME/

# Upload SQL files
gsutil cp coldsense_schema.sql gs://$BUCKET_NAME/
gsutil cp coldsense_data.sql gs://$BUCKET_NAME/

# Grant Cloud SQL access
$SERVICE_ACCOUNT = gcloud sql instances describe $INSTANCE_NAME --project=$PROJECT_ID --format="get(serviceAccountEmailAddress)"
gsutil iam ch serviceAccount:${SERVICE_ACCOUNT}:objectViewer gs://$BUCKET_NAME

# Import schema
gcloud sql import sql $INSTANCE_NAME gs://$BUCKET_NAME/coldsense_schema.sql --database=coldsense --project=$PROJECT_ID

# Wait 5-10 minutes...

# Import data
gcloud sql import sql $INSTANCE_NAME gs://$BUCKET_NAME/coldsense_data.sql --database=coldsense --project=$PROJECT_ID

# Wait 10-20 minutes...

═══════════════════════════════════════════════════════════════════════════════
STEP 6: VERIFY DATA (5 minutes)
═══════════════════════════════════════════════════════════════════════════════

# SSH to VM
gcloud compute ssh coldsense-backend-vm --zone=asia-south1-a

# === ON THE VM ===

# Connect to Cloud SQL (use the DATABASE_URL from cloud-sql-credentials.txt)
psql "postgresql://coldsense_app:YOUR_PASSWORD@CLOUD_SQL_IP:5432/coldsense?sslmode=require"

# Check tables
\dt

# Count records
SELECT 'users' as table_name, COUNT(*) as count FROM users
UNION ALL
SELECT 'facilities', COUNT(*) FROM facilities
UNION ALL
SELECT 'sensor_readings', COUNT(*) FROM sensor_readings
UNION ALL
SELECT 'products', COUNT(*) FROM products;

# Exit
\q
exit

═══════════════════════════════════════════════════════════════════════════════
STEP 7: UPDATE BACKEND (10 minutes)
═══════════════════════════════════════════════════════════════════════════════

# SSH to VM
gcloud compute ssh coldsense-backend-vm --zone=asia-south1-a

# === ON THE VM ===

cd ~/coldsense/backend

# Backup current .env
cp .env .env.backup

# Edit .env
nano .env

# UPDATE THIS LINE:
# OLD: DATABASE_URL=postgresql://postgres.mmxfqodybvdlhpjbcdnq:...@aws-0-ap-south-1.pooler.supabase.com:6543/postgres
# NEW: DATABASE_URL=postgresql://coldsense_app:YOUR_PASSWORD@CLOUD_SQL_IP:5432/coldsense?sslmode=require

# KEEP THESE (for Supabase Auth):
# SUPABASE_URL=https://mmxfqodybvdlhpjbcdnq.supabase.co
# SUPABASE_KEY=your_anon_key

# Save and exit: Ctrl+O, Enter, Ctrl+X

# Restart backend
cd ~/coldsense
docker-compose down
docker-compose up -d

# Check logs
docker-compose logs -f backend

# Exit VM
exit
