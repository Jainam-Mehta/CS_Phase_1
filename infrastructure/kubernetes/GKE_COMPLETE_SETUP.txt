STEP 12: GET LOAD BALANCER IP (5 minutes)
═══════════════════════════════════════════════════════════════════════════════

Write-Host "`n⏳ Waiting for LoadBalancer to assign external IP..." -ForegroundColor Yellow
Write-Host "This can take 2-5 minutes..." -ForegroundColor Yellow

# Wait for external IP
$EXTERNAL_IP = ""
$attempts = 0
while (($EXTERNAL_IP -eq "" -or $EXTERNAL_IP -eq "<pending>") -and $attempts -lt 60) {
    Start-Sleep -Seconds 10
    $EXTERNAL_IP = kubectl get svc backend-service --namespace=coldsense -o jsonpath='{.status.loadBalancer.ingress[0].ip}' 2>$null
    $attempts++
    Write-Host "." -NoNewline -ForegroundColor Yellow
}

Write-Host ""
if ($EXTERNAL_IP -ne "" -and $EXTERNAL_IP -ne "<pending>") {
    Write-Host "`n✅ LoadBalancer IP assigned!" -ForegroundColor Green
    Write-Host "External IP: $EXTERNAL_IP" -ForegroundColor Cyan
    Write-Host "API URL: http://${EXTERNAL_IP}:8000" -ForegroundColor Cyan
    Write-Output "`n=== LOAD BALANCER ===" | Out-File -Append "kubernetes-credentials.txt"
    Write-Output "External IP: $EXTERNAL_IP" | Out-File -Append "kubernetes-credentials.txt"
    Write-Output "API URL: http://${EXTERNAL_IP}:8000" | Out-File -Append "kubernetes-credentials.txt"
} else {
    Write-Host "`n⚠️ LoadBalancer IP not assigned yet. Check with:" -ForegroundColor Yellow
    Write-Host "kubectl get svc backend-service --namespace=coldsense" -ForegroundColor White
}
