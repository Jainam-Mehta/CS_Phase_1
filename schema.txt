-- Trigger to auto-generate site_code on insert
CREATE OR REPLACE FUNCTION public.generate_site_code()
RETURNS TRIGGER AS $$
DECLARE
  next_num INTEGER;
BEGIN
  IF NEW.site_code IS NULL THEN
    SELECT COALESCE(MAX(CAST(SUBSTRING(site_code, 2) AS INTEGER)), 0) + 1 INTO next_num
    FROM public.sites
    WHERE owner_profile_id = NEW.owner_profile_id;
    NEW.site_code := 'S' || LPAD(next_num::text, 3, '0');
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sites_auto_site_code
BEFORE INSERT ON public.sites
FOR EACH ROW
EXECUTE FUNCTION public.generate_site_code();

-- ============================================================================
-- TRIGGERS FOR NET WORTH AUTO-MAINTENANCE (Using SECURITY DEFINER to bypass RLS)
-- ============================================================================

-- Trigger Function 1: Update net worth when batch is created/updated/deleted
CREATE OR REPLACE FUNCTION public.update_farmer_net_worth_on_batch_change()
RETURNS TRIGGER AS $$
DECLARE
  farmer_id_var uuid;
  total_when_stored numeric(14, 2);
  total_crates numeric;
  total_kg numeric(12, 2);
  batch_count integer;
BEGIN
  IF TG_OP = 'INSERT' THEN
    farmer_id_var := NEW.farmer_id;
  ELSIF TG_OP = 'UPDATE' THEN
    farmer_id_var := NEW.farmer_id;
  ELSIF TG_OP = 'DELETE' THEN
    farmer_id_var := OLD.farmer_id;
  END IF;

  SELECT
    COALESCE(SUM(initial_quantity_kg * snapshot_price_per_kg), 0),
    COALESCE(SUM(initial_quantity_kg / 25), 0),
    COALESCE(SUM(initial_quantity_kg), 0),
    COUNT(*)
  INTO
    total_when_stored, total_crates, total_kg, batch_count
  FROM public.batches
  WHERE farmer_id = farmer_id_var AND remaining_quantity_kg > 0;

  INSERT INTO public.farmer_inventory_net_worth (farmer_id, total_inventory_value_when_stored, total_active_batches, total_crates_in_storage, total_quantity_kg, last_updated)
  VALUES (farmer_id_var, total_when_stored, batch_count, COALESCE(total_crates, 0), total_kg, CURRENT_TIMESTAMP)
  ON CONFLICT (farmer_id) DO UPDATE SET
    total_inventory_value_when_stored = total_when_stored,
    total_active_batches = batch_count,
    total_crates_in_storage = COALESCE(total_crates, 0),
    total_quantity_kg = total_kg,
    last_updated = CURRENT_TIMESTAMP;

  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER SET search_path = public;

CREATE TRIGGER trg_batches_update_net_worth
AFTER INSERT OR UPDATE OR DELETE ON public.batches
FOR EACH ROW
EXECUTE FUNCTION public.update_farmer_net_worth_on_batch_change();

-- Trigger Function 2: Update net worth when batch allocation is removed (soft delete)
CREATE OR REPLACE FUNCTION public.update_farmer_net_worth_on_allocation_remove()
RETURNS TRIGGER AS $$
DECLARE
  farmer_id_var uuid;
  total_when_stored numeric(14, 2);
  total_crates numeric;
  total_kg numeric(12, 2);
  batch_count integer;
BEGIN
  SELECT farmer_id INTO farmer_id_var FROM public.batches WHERE id = NEW.batch_id;

  SELECT
    COALESCE(SUM(initial_quantity_kg * snapshot_price_per_kg), 0),
    COALESCE(SUM(initial_quantity_kg / 25), 0),
    COALESCE(SUM(initial_quantity_kg), 0),
    COUNT(*)
  INTO
    total_when_stored, total_crates, total_kg, batch_count
  FROM public.batches
  WHERE farmer_id = farmer_id_var AND remaining_quantity_kg > 0;

  UPDATE public.farmer_inventory_net_worth
  SET
    total_inventory_value_when_stored = total_when_stored,
    total_active_batches = batch_count,
    total_crates_in_storage = COALESCE(total_crates, 0),
    total_quantity_kg = total_kg,
    last_updated = CURRENT_TIMESTAMP
  WHERE farmer_id = farmer_id_var;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER SET search_path = public;

CREATE TRIGGER trg_allocations_update_net_worth
AFTER UPDATE ON public.batch_room_allocations
FOR EACH ROW
WHEN (OLD.removed_at IS NULL AND NEW.removed_at IS NOT NULL)
EXECUTE FUNCTION public.update_farmer_net_worth_on_allocation_remove();

-- 33. FARMER_PAYMENT_HISTORY (Immutable Record of All Payments Made by Farmers)
-- Purpose: Permanent audit trail for payment history (for insights/alerts)
-- Workflow: On payment completion, record inserted here, then farmer_daily_storage_costs rows deleted
CREATE TABLE public.farmer_payment_history (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  farmer_id uuid NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  owner_id uuid NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
  room_id uuid NOT NULL REFERENCES public.cold_storage_rooms(id) ON DELETE CASCADE,
  site_id uuid NOT NULL REFERENCES public.sites(id) ON DELETE CASCADE,
  
  -- Payment Period Details
  period_start_date date NOT NULL,                -- First day of billing period
  period_end_date date NOT NULL,                  -- Last day of billing period (payment day)
  
  -- Summary Statistics
  total_daily_entries integer NOT NULL,           -- # of days in period
  total_crates_stored numeric(12, 2),             -- Total crates summed across period
  total_quantity_kg numeric(12, 2),               -- Total kg summed across period (crates × 25kg)
  
  -- Payment Amount (cumulative_cost at time of payment)
  amount_paid numeric(14, 2) NOT NULL,            -- Total amount farmer paid to owner
  
  -- Payment Metadata
  payment_date timestamp with time zone NOT NULL DEFAULT CURRENT_TIMESTAMP,  -- When payment recorded
  payment_received_by_owner_id uuid REFERENCES public.profiles(id) ON DELETE SET NULL,  -- Which owner recorded it
  
  payment_method character varying(50),           -- 'cash', 'bank_transfer', 'upi', 'check', etc.
  transaction_id character varying(100),          -- Bank transaction ID or reference number
  remarks text,                                   -- Any additional notes
  
  created_at timestamp with time zone DEFAULT CURRENT_TIMESTAMP,
  
  CONSTRAINT amount_positive CHECK (amount_paid > 0)
);

-- Indexes for fast queries (Insights/Alerts tabs)
CREATE INDEX idx_farmer_payment_history_farmer ON public.farmer_payment_history(farmer_id);
CREATE INDEX idx_farmer_payment_history_owner ON public.farmer_payment_history(owner_id);
CREATE INDEX idx_farmer_payment_history_room ON public.farmer_payment_history(room_id);
CREATE INDEX idx_farmer_payment_history_site ON public.farmer_payment_history(site_id);
CREATE INDEX idx_farmer_payment_history_payment_date ON public.farmer_payment_history(payment_date DESC);
CREATE INDEX idx_farmer_payment_history_period ON public.farmer_payment_history(farmer_id, room_id, period_end_date DESC);
