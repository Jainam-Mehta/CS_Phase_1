
# ==========================================================================================
# NEW WAY: SYSTEMD AUTO-START SERVICE (RECOMMENDED - Updated 2026-08-26)
# ==========================================================================================
# This is the new recommended approach. The backend now runs as a systemd service and
# automatically starts when the VM boots. No manual Docker Compose commands needed!
#
# Benefits:
# ✅ Auto-starts on VM reboot (no manual intervention)
# ✅ Lower overhead than Docker
# ✅ Faster startup (~5-10 seconds vs 30+ with Docker)
# ✅ Better logging and monitoring
# ✅ Automatic restart on failure
#
# ==========================

# STARTING THE GCP VM (with systemd auto-start backend)
# =====================================================

# 1. Start the VM from local machine
gcloud compute instances start coldsense-production-vm --zone=asia-south1-c

# 2. Wait ~1-2 minutes for VM to fully boot

# 3. SSH into the VM (optional - just to verify)
gcloud compute ssh coldsense-production-vm --zone=asia-south1-c

# 4. On the VM, verify backend is running (systemd auto-started it)
sudo systemctl status coldsense-backend

# 5. Verify API is responsive
curl http://localhost:8000/health

# 6. Check backend logs
sudo journalctl -u coldsense-backend -n 50

# 7. Verify MQTT broker is running (still in Docker)
docker ps | grep mqtt-broker

# 8. Exit SSH when done
exit

# STOPPING THE GCP VM (systemd service stops automatically)
# ===========================================================

# 1. Stop the VM from local machine
gcloud compute instances stop coldsense-production-vm --zone=asia-south1-c

# That's it! The systemd service will stop when the VM shuts down.
# No need to SSH and manually stop Docker containers.

# ==========================
# USEFUL COMMANDS (with systemd service)
# ==========================

# Check backend service status
sudo systemctl status coldsense-backend

# View real-time backend logs
sudo journalctl -u coldsense-backend -f

# View last 50 lines of backend logs
sudo journalctl -u coldsense-backend -n 50

# Check if backend is running
sudo systemctl is-active coldsense-backend

# Manually restart the backend (if needed)
sudo systemctl restart coldsense-backend

# Stop the backend manually (if needed)
sudo systemctl stop coldsense-backend

# Start the backend manually (if needed)
sudo systemctl start coldsense-backend

# View memory and CPU usage of backend process
ps aux | grep python

# Check if API endpoint is responding
curl http://localhost:8000/health
curl http://localhost:8000/docs
