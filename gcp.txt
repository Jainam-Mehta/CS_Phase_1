# ==========================
# STARTING GCP SERVICES
# ==========================

# 1. Start the VM (from local machine)
gcloud compute instances start coldsense-production-vm --zone=asia-south1-c

# 2. Wait ~1-2 minutes for VM to fully boot, then SSH into it
gcloud compute ssh coldsense-production-vm --zone=asia-south1-c

# 3. On the VM, pull latest code from GitHub
cd /home/jainammehta250
if [ -d CS_Project ]; then
  cd CS_Project && git pull
else
  git clone https://github.com/Jainam-Mehta/CS_Phase_1.git CS_Project
fi

# 4. Navigate to infrastructure folder
cd /home/jainammehta250/CS_Project/infrastructure

# 5. Pull latest Docker images and start all services
docker compose pull
docker compose up -d --force-recreate

# 6. Verify all containers are running
docker ps

# 7. Check backend health
docker logs coldsense-backend --tail=20

# 8. Check MQTT subscriber
docker logs coldsense-mqtt --tail=20

# 9. Exit SSH when done monitoring
exit

# ==========================
# STOPPING GCP SERVICES
# ==========================

# 1. SSH into the VM (from local machine)
gcloud compute ssh coldsense-production-vm --zone=asia-south1-c

# 2. On the VM, navigate to infrastructure folder
cd /home/jainammehta250/CS_Project/infrastructure

# 3. Stop all Docker containers
docker compose down

# 4. Verify all containers stopped
docker ps

# 5. Exit SSH session
exit

# 6. Stop the VM instance (from local machine)
gcloud compute instances stop coldsense-production-vm --zone=asia-south1-c

# ==========================
# USEFUL COMMANDS (ON VM)
# ==========================

# View all running containers
docker ps

# View all containers (running + stopped)
docker ps -a

# View container logs in real-time
docker logs -f coldsense-backend
docker logs -f coldsense-mqtt
docker logs -f coldsense-postgres

# Check services status
docker compose ps

# Restart a specific service
docker compose restart coldsense-backend
